```typescript
test("좌표 없으면 순수 검색 URL", () => {

  assert.equal(url, "https://map.naver.com/p/search/" + encodeURIComponent("시각장애인 안마원"));

  const url = buildNaverMapUrl("강남구 주민센터");
  assert.ok(!url.includes(" "));

test("앱 스킴은 navermap://search?query=", () => {
  const url = buildNaverAppUrl("시각장애인 안마원");
  assert.ok(url.startsWith("navermap://search?query="));
  assert.ok(url.includes(encodeURIComponent("시각장애인 안마원")));
});
```

- [ ] **Step 3: 최소 구현**

`anma-hub/lib/naver.ts` (Task 2 SPIKE 결과의 좌표 포맷을 반영 — 아래는 기본 가정 포맷):
```typescript
export type Coords = { lat: number; lng: number };

const BASE = "https://map.naver.com/p/search/";

/** 네이버 지도 검색 URL. 좌표가 있으면 지도 중심을 사용자 위치로 이동시킨다. */
export function buildNaverMapUrl(query: string, coords?: Coords): string {
  const path = BASE + encodeURIComponent(query);
  if (!coords) return path;
  // c=경도,위도,zoom,0,0,0,dh (SPIKE로 확정한 포맷)
  return `${path}?c=${coords.lng},${coords.lat},15,0,0,0,dh`;
}

/** 모바일 네이버 지도 앱 스킴. 앱 미설치 시 호출측에서 웹으로 폴백. */
export function buildNaverAppUrl(query: string): string {
  return `navermap://search?query=${encodeURIComponent(query)}&appname=anma.zucca100.com`;
}
```

```tsx
"use client";
import { useState } from "react";
import { MapPin, Navigation, Loader2 } from "lucide-react";
import { buildNaverMapUrl } from "@/lib/naver";
import { getSidos } from "@/lib/regions";

export function LocationFinder({ query, label }: { query: string; label: string }) {
  const [status, setStatus] = useState<"idle" | "locating" | "denied">("idle");
  const sidos = getSidos();

  function openNearby() {
    if (!("geolocation" in navigator)) { setStatus("denied"); return; }
    setStatus("locating");
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        const url = buildNaverMapUrl(query, {
          lat: pos.coords.latitude,
          lng: pos.coords.longitude,
        });
        setStatus("idle");
        window.location.href = url; // 같은 탭 이동(포트폴리오 관례: 지도 링크 _blank 미적용)
      },
      () => setStatus("denied"),
      { enableHighAccuracy: true, timeout: 8000 }
    );
  }

  return (
    <div className="tile flex flex-col gap-3 px-4 py-4">
      <button
        type="button"
        onClick={openNearby}
        disabled={status === "locating"}
        className="inline-flex items-center justify-center gap-2 rounded-xl bg-accent px-4 py-3 text-[15px] font-bold text-on-primary hover:opacity-90 disabled:opacity-60"
      >
        {status === "locating" ? <Loader2 size={16} className="animate-spin" /> : <Navigation size={16} />}
        내 위치로 {label} 찾기
      </button>

      {status === "denied" && (
        <p className="text-[13px] leading-[1.6] text-muted">
          위치 접근을 사용할 수 없어요. 아래에서 지역을 선택하세요.
        </p>
      )}

      <div>
        <p className="mb-2 flex items-center gap-1.5 text-[13px] font-semibold text-muted">
          <MapPin size={13} className="text-accent" /> 지역으로 찾기
        </p>
        <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
          {sidos.map((s) => (
            <a
              key={s.slug}
              href={buildNaverMapUrl(`${s.name} ${query}`)}
              className="rounded-lg bg-band px-3 py-2 text-center text-[13.5px] font-semibold text-foreground hover:text-accent"
            >
              {s.name}
            </a>
          ))}
        </div>
      </div>
    </div>
  );
}
```

```typescript
import type { MetadataRoute } from "next";
import { SITE } from "@/lib/site";
import { getTargets } from "@/lib/targets";
import { getGuides } from "@/lib/guides";
import { getSidos } from "@/lib/regions";

export default function sitemap(): MetadataRoute.Sitemap {
  const base = SITE.url;
  const urls = [
    "", "/find", "/related/maternity",
    ...getTargets().map((t) => `/target/${t.slug}`),
    ...getGuides().map((g) => `/guide/${g.slug}`),
    ...getSidos().map((s) => `/find/${s.slug}`),
  ];
  const now = new Date();
  return urls.map((u) => ({ url: `${base}${u}`, lastModified: now, changeFrequency: "monthly", priority: u === "" ? 1 : 0.7 }));
}
```
